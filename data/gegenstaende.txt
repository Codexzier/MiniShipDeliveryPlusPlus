# unter "waren" benutzt wird.
#   name         = Anzeigename (Tooltip)
#   beschreibung = kurze Beschreibung (Tooltip)

#   effekt       = keiner, ausdauer, leben, enterhaken, max_ausdauer

#   verbrauchbar = 1, wenn der Gegenstand beim Benutzen verschwindet
#   einzigartig  = 1, wenn man ihn nur einmal besitzen kann
#   stapel_max   = wie viele Stück in ein Inventarfeld passen
# ============================================================
