# ============================================================
# Die Seekarte
#
# [Welt]      Größe der Karte in Kacheln, Startwert für den Zufall, Startinsel
# [Insel_*]   Eine Insel
#   name        = Anzeigename
#   typ         = stadt, dorf oder wild
#   landschaft  = palmen, wald, stadt oder grusel (Bewuchs)
#   x, y        = Mittelpunkt in Kacheln
#   radius      = Größe der Insel in Kacheln
#   hafen       = Richtung des einzigen Hafens: nord, ost, sued oder west
#   gebaeude    = kontor, museum, werft, taverne, laden:<Händler>, hersteller:<Hersteller>
#   markt       = Händler mit Marktstand auf dem Platz (höchstens drei)
#   produziert  = Waren, die hier billig sind (Kontor verkauft günstig)
#   bedarf      = Waren, die hier gesucht sind (Kontor zahlt mehr)
# [Riff_*]    Flaches Wasser mit Felsen (Schiffe mit großem Tiefgang laufen auf)
# [Piraten_*] Piratengebiet (staerke = wie viele Piratenschiffe gleichzeitig)
# [Monster_*] Gebiet, in dem nachts Seeungeheuer jagen
# [Route_*]   Sichere Handelsroute (Bojen; Piraten greifen dort selten an)
#   von, nach  = Inseln
#   ueber      = Zwischenpunkte "x y", durch Komma getrennt
# ============================================================
