# ============================================================
# Schiffsklassen (Kauf in der Werft)
#
# name             = Anzeigename
# modell           = Modell aus data/modelle.txt
# beschreibung     = Kurztext
# preis            = Kaufpreis in Credits (das alte Schiff wird zur Hälfte angerechnet)
# laderaum_breite  = Felder nebeneinander im Laderaum (Minispiel)
# laderaum_hoehe   = Reihen im Laderaum
# geschwindigkeit  = Höchstgeschwindigkeit in Kacheln pro Sekunde
# reparatur        = Rumpfpunkte, die die Crew pro Spielstunde auf See flickt
# panzerung        = Schadensminderung in Prozent
# abwehr           = Kanonenschaden gegen Piraten und Ungeheuer
# wendigkeit       = Drehgeschwindigkeit in Grad pro Sekunde
# sichtweite       = Scanner / Ausguck in Kacheln (Artefakte, Gefahren)
# handel           = Handelsfaktor in Prozent (100 = normal)
# rumpf            = Rumpfpunkte (Gesundheit)
# antrieb          = wind, motor oder hybrid (Segel + Maschine)
# treibstoff_max   = Tankgröße in Einheiten (nur motor/hybrid)
# verbrauch        = Treibstoff pro Sekunde bei voller Maschinenfahrt
# tiefgang         = Tiefgang in Metern (flacheres Wasser = Grundberührung)
# mindest_crew     = Mindestbesatzung (darunter gibt es Abzüge)
# max_crew         = Kojen an Bord
# ============================================================
