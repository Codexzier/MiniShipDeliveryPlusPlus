# ============================================================
# Händlerarten (Marktstand auf dem Platz oder eigener Laden)
#
# name                = Anzeigename
# art                 = Warenkategorie, mit der gehandelt wird: gemuese, obst, fertigware ...
# kauft               = feste Liste der angekauften Waren (leer = Bedarf der Insel in dieser Kategorie,
#                       sonst alle Waren der Kategorie, die die Insel nicht selbst herstellt)
# verkauft            = feste Liste der verkauften Waren (leer = eigene Waren der Insel in dieser Kategorie)
# figur               = Figur vor dem Stand
# bedarf_max          = Höchstmenge, die je Ware angekauft wird
# bedarf_modus        = taeglich (Bedarf jeden Morgen voll) oder steigend (wächst langsam)
# bedarf_anstieg      = Zuwachs des Bedarfs je Spielstunde (nur steigend)
# bestand_max         = Höchstbestand je verkaufter Ware
# bestand_erholung    = Wiederauffüllung je Spielstunde
# preisfaktor_ankauf  = Händler zahlt so viel mal den Inselpreis (mehr als das Kontor!)
# preisfaktor_verkauf = Händler verlangt so viel mal den Inselpreis
# frische_bonus       = Zuschlag für ganz frische verderbliche Ware (0.3 = +30 %)
# lieferbonus         = Jede gelieferte Einheit füllt den Verkaufsbestand um so viel auf
# ============================================================
